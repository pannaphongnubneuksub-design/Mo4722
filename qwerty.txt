        <div class="upload-section">
            <div class="upload-btn-wrapper">
                <button class="btn">เลือกรูปภาพเพื่อเพิ่มลงแกลเลอรี</button>
                <!-- accept="image/*" คือยอมรับเฉพาะไฟล์รูปภาพ, multiple คือเลือกหลายรูปพร้อมกันได้ -->

            </div>
            <p class="upload-desc">รองรับไฟล์ JPG, PNG, GIF (สามารถเลือกหลายรูปพร้อมกันได้)</p>
        </div>

        <!-- ส่วนแสดงแกลเลอรี -->
        <div class="gallery" id="imageGallery">
            <!-- รูปภาพที่ถูกเลือกจะมาแสดงตรงนี้ -->
        </div>
    </div>

    <script>
        // ดึง Element จาก HTML
        const imageInput = document.getElementById('imageInput');
        const imageGallery = document.getElementById('imageGallery');

        // เมื่อมีการเลือกไฟล์
        imageInput.addEventListener('change', function(event) {
            const files = event.target.files;

            if (files && files.length > 0) {
                // วนลูปอ่านไฟล์ที่เลือกทั้งหมด
                for (let i = 0; i < files.length; i++) {
                    const file = files[i];

                    // ตรวจสอบว่าเป็นไฟล์รูปภาพหรือไม่
                    if (file.type.startsWith('image/')) {
                        const reader = new FileReader();

                        // เมื่ออ่านไฟล์เสร็จเรียบร้อย
                        reader.onload = function(e) {
                            // สร้าง div สำหรับกรอบรูปภาพ
                            const galleryItem = document.createElement('div');
                            galleryItem.className = 'gallery-item';

                            // สร้าง element <img>
                            const img = document.createElement('img');
                            img.src = e.target.result; // กำหนด source ให้เป็น Data URL ของรูป
                            img.alt = file.name;

                            // นำ <img> ใส่เข้าไปใน div
                            galleryItem.appendChild(img);

                            // นำ div ใส่เข้าไปในแกลเลอรี (ไว้ด้านหน้าสุดด้วย prepend หรือด้านหลังด้วย appendChild)
                            imageGallery.prepend(galleryItem);
                        };

                        // อ่านไฟล์รูปภาพให้เป็น Data URL
                        reader.readAsDataURL(file);
                    }
                }
            }
        });
    </script>
